  zh ? '# 综合研究笔记\n\n本地合成回答 · 未连接模型服务。' : '# Research synthesis\n\nLocal synthetic response · No model service is connected.',

const failures = new Set<string>();
export const syntheticModel: ChatModelAdapter = {
  async *run({messages, abortSignal}) {
    const user = [...messages].reverse().find(m=>m.role==='user');
    const query = user?.content.filter(p=>p.type==='text').map(p=>(p as {text:string}).text).join(' ') || '';
    const shouldFail = /\[fail\]/i.test(query) && !failures.has(query);
    for(let end=0; end<longResponse.length; end+=72){
      if(abortSignal.aborted) return;
      await new Promise<void>(resolve=>setTimeout(resolve,60));
      if(abortSignal.aborted) return;
      if(shouldFail && end>=432){ failures.add(query); throw new Error(zh ? '合成服务暂时不可用。重试会继续使用本地示例。' : 'The synthetic response failed once. Retry uses the same local example.'); }
      yield {content:[{type:'text',text:longResponse.slice(0,end+72)}]};
    }
  },
};

  if (!src) return children;

  return (
    <Dialog>

const AttachmentUI: FC = () => {
  const aui = useAui();
  const [filenameOpen, setFilenameOpen] = useState(false);
  const {
    isComposer,
    src,
    isImage,
    name,
    typeLabel,
    isUploading,
    isError,
    errorMessage,
  } = useAttachmentState();

  // Source-backed filename chips for metadata-only attachments. Keep image previews unchanged.
  if (!src) {
    const zh = document.documentElement.lang.startsWith("zh");
    return (
      <AttachmentPrimitive.Root>
        <Tooltip open={filenameOpen} onOpenChange={setFilenameOpen}>
          <TooltipTrigger render={
            <ComposerAttachmentChip
              tabIndex={0}
              className="outline-none focus-visible:ring-1 focus-visible:ring-ring/50"
              onFocus={() => setFilenameOpen(true)}
              onBlur={() => setFilenameOpen(false)}
              onClick={() => setFilenameOpen(true)}
              title={name}
              aria-label={`${name} · ${zh ? "仅文件名" : "Filename only"}`}
              attachment={{ name, meta: errorMessage ?? (zh ? "仅文件名" : "Filename only"), state: isError ? "error" : isUploading ? "uploading" : "done" }}
              onRemove={isComposer ? () => {
                aui.attachment.remove();
                requestAnimationFrame(() => document.querySelector<HTMLTextAreaElement>(".aui-composer-input")?.focus());
              } : undefined}
            />
          } />
          <TooltipContent side="top" nonInteractive role="tooltip" className="pointer-events-none flex-col items-start wrap-anywhere">
            <AttachmentPrimitive.Name />
            <p>{zh ? "保留文件名；不读取或发送内容。" : "Filename retained; contents are not read or sent."}</p>
          </TooltipContent>
        </Tooltip>
      </AttachmentPrimitive.Root>
    );
  }

  return (
    <TooltipProvider>
      <Tooltip>
        <AttachmentPrimitive.Root
          className={cn(
            "aui-attachment-root relative",
            isComposer &&
              "animate-in fade-in-0 zoom-in-95 duration-200 motion-reduce:animate-none",
            isImage &&
              !isComposer &&
              "aui-attachment-root-message only:*:first:size-24",
          )}
        >
          <AttachmentPreviewDialog src={src} name={name}>
            <TooltipTrigger
              render={
                <div
                  className={cn(
                    "aui-attachment-tile bg-muted hover:after:bg-foreground/10 focus-visible:ring-ring/50 relative size-14 overflow-hidden rounded-[calc(var(--composer-radius,1rem)-var(--composer-padding,8px))] transition-transform outline-none after:pointer-events-none after:absolute after:inset-0 after:rounded-[inherit] after:ring-1 after:ring-black/10 after:transition-colors after:ring-inset focus-visible:ring-1 motion-reduce:transition-none dark:after:ring-white/10",
                    src
                      ? "cursor-zoom-in active:scale-[0.96]"
                      : "cursor-default",
                    isError &&
                      "after:ring-destructive/60 dark:after:ring-destructive/60",
                  )}
                  role={src ? "button" : "group"}
                  tabIndex={0}
                  aria-label={`${src ? `Preview ${name}` : `${typeLabel} attachment ${name}`}${
                    isError
                      ? ", upload failed"
                      : isUploading
                        ? ", uploading"
                        : ""
                  }`}
                />
              }
            >
              <AttachmentThumb src={src} />

                <div
                  aria-hidden="true"
                  className="aui-attachment-tile-uploading bg-background/60 animate-in fade-in-0 absolute inset-0 flex items-center justify-center backdrop-blur-[2px] motion-reduce:animate-none"
