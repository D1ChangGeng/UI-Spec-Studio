"""Local orchestration; never edits an active participant application."""
from pathlib import Path
from datetime import datetime, timezone
import argparse, hashlib, json, shutil, zipfile

ROOT = Path('/workspace/scratch/43b440780c87')
REPO = ROOT / 'ui-spec-studio'
EVENTS = REPO / 'docs/experiments/adaptive-granularity/run-events.json'

def now():
    return datetime.now(timezone.utc).isoformat()

def event(cell, action, **extras):
    data = json.loads(EVENTS.read_text())
    data['events'].append({'cell': cell, 'event': action, 'observed_at_utc': now(), **extras})
    EVENTS.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')

def snapshot(cell, stage):
    src = ROOT / 'granularity-cells' / cell
    dst = ROOT / 'granularity-archives' / cell / stage
    dst.mkdir(parents=True)
    files = {}
    for path in sorted(src.rglob('*')):
        if not path.is_file():
            continue
        rel = path.relative_to(src)
        if any(part in ('ui-spec', 'node_modules', '__pycache__', '.git') for part in rel.parts):
            continue
        if path.suffix == '.pyc' or path.name in ('common-guidance.md', 'common-guidance-raw.md'):
            continue
        if path.is_symlink():
            raise ValueError(f'Refuse unreviewed symlink: {rel}')
        out = dst / rel
        out.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(path, out)
        files[str(rel)] = hashlib.sha256(path.read_bytes()).hexdigest()
    identity = {'cell': cell, 'stage': stage, 'observed_at_utc': now(), 'files': files}
    (dst / 'snapshot-identity.json').write_text(json.dumps(identity, indent=2) + '\n')
    event(cell, f'{stage}_completed_and_frozen', files=len(files))
    if stage == 'initial':
        shutil.copy2(ROOT / 'granularity-cells/continuations' / cell / 'TASK.md', src / 'CONTINUE.md')
    print(json.dumps({'cell': cell, 'stage': stage, 'files': len(files)}))

def archive(cell):
    """Retain exact bytes once, mapping original paths to portable archive paths."""
    src = ROOT / 'granularity-archives' / cell
    dst = REPO / 'docs/experiments/adaptive-granularity/runs' / cell
    dst.mkdir(parents=True)
    identities = {stage: json.loads((src/stage/'snapshot-identity.json').read_text())
                  for stage in ('initial', 'final')}
    by_hash = {}
    index = {'cell': cell, 'scope': 'Exact participant files; Markdown is retained as text to preserve original local links.',
             'snapshots': {}}
    for stage in ('final', 'initial'):
        snapshot_index = {}
        for name, digest in identities[stage]['files'].items():
            original = src/stage/name
            if hashlib.sha256(original.read_bytes()).hexdigest() != digest:
                raise ValueError(f'Snapshot changed: {cell}/{stage}/{name}')
            if stage == 'initial' and digest in by_hash:
                target = by_hash[digest]
            else:
                target = Path(stage)/name
                if original.suffix == '.md':
                    target = target.with_suffix('.md.original.txt')
                out = dst/target
                out.parent.mkdir(parents=True, exist_ok=True)
                shutil.copy2(original, out)
                by_hash[digest] = target
            snapshot_index[name] = {'path': str(target), 'sha256': digest}
        index['snapshots'][stage] = {'observed_at_utc': identities[stage]['observed_at_utc'], 'files': snapshot_index}
    # Keep raw captures without flooding the source tree with near-identical frames.
    # Application assets remain directly runnable; every capture is byte-addressable.
    captures = sorted(p for p in dst.rglob('*') if p.is_file()
                      and p.suffix in ('.png', '.webm', '.mp4')
                      and 'app' not in p.relative_to(dst).parts)
    batches, batch, total = [], [], 0
    for capture in captures:
        size = capture.stat().st_size
        if batch and total + size > 4 * 1024 * 1024:
            batches.append(batch); batch, total = [], 0
        batch.append(capture); total += size
    if batch:
        batches.append(batch)
    remap, bundles = {}, []
    for number, paths in enumerate(batches, 1):
        bundle = f'captures-{number:02d}.zip'
        with zipfile.ZipFile(dst/bundle, 'w', compression=zipfile.ZIP_DEFLATED, compresslevel=9) as z:
            for capture in paths:
                name = str(capture.relative_to(dst))
                info = zipfile.ZipInfo(name, (1980, 1, 1, 0, 0, 0))
                info.compress_type = zipfile.ZIP_DEFLATED
                info.external_attr = 0o100644 << 16
                z.writestr(info, capture.read_bytes())
                remap[name] = bundle
        bundles.append({'path': bundle, 'sha256': hashlib.sha256((dst/bundle).read_bytes()).hexdigest(),
                        'members': len(paths)})
    for stage in index['snapshots'].values():
        for item in stage['files'].values():
            if item['path'] in remap:
                item['member'] = item['path']
                item['path'] = remap[item['path']]
    for capture in captures:
        capture.unlink()
    index['capture_bundles'] = bundles
    (dst/'archive-index.json').write_text(json.dumps(index, ensure_ascii=False, indent=2)+'\n')
    print(json.dumps({'cell': cell, 'unique_files': len(by_hash), 'file_observations': sum(len(x['files']) for x in index['snapshots'].values())}))

parser = argparse.ArgumentParser()
parser.add_argument('action', choices=['initial', 'final', 'archive', 'event'])
parser.add_argument('cell')
parser.add_argument('--event-name')
args = parser.parse_args()
if args.action in ('initial', 'final'):
    snapshot(args.cell, args.action)
elif args.action == 'archive':
    archive(args.cell)
else:
    event(args.cell, args.event_name)
